      Install

      pnpm add @mark1russell7/optional
      import { fromNullable, getOr, map, some, type Optional } from "@mark1russell7/optional";

const width: Optional<number> = some(4);
getOr(map(width, (w) => w * 2), 0); // 8
getOr(fromNullable(undefined), 0);  // 0
